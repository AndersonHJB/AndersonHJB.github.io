import numpy as np

# 定义晶格类
class Lattice:
    # 初始化方法
    def __init__(self):
        # 用一个列表来存储所有的原子位置
        self.atoms = []

    # 方法用于生成并存储原子的位置
    def populate_atoms(self):
        # 通过嵌套循环确定原子的x和y位置
        for x in np.arange(-2, 3):
            for y in np.arange(-2, 3):
                # 计算并添加原子的位置到atoms列表
                self.atoms.extend([
                    [4 * x, 2 * y * np.sqrt(3)],
                    [2 + 4 * x, 2 * y * np.sqrt(3)],
                    [1 + 4 * x, np.sqrt(3) + 2 * y * np.sqrt(3)],
                    [3 + 4 * x, np.sqrt(3) + 2 * y * np.sqrt(3)]
                ])

    # 用于在图上绘制原子的方法
    def plot_atoms(self, ax):
        # 遍历每个原子的位置并在图上绘制
        for atom in self.atoms:
            ax.plot(atom[0], atom[1], "yo")

    # 用于在图上绘制圆形的方法
    def plot_circle(self, ax, radius, color="r", fill=False):
        # 创建一个圆形并添加到图上
        circle = plt.Circle(xy=(0, 0), radius=radius, color=color, fill=fill)
        ax.add_patch(circle)

    # 可视化方法
    def visualize(self):
        # 从原子位置中提取x和y的值
        x_vals, y_vals = zip(*self.atoms)
        # 计算x和y的范围
        x_dim = max(x_vals) - min(x_vals)
        y_dim = max(y_vals) - min(y_vals)
        
        # 创建一个图形

        self.plot_circle(ax, 2, "r")
        self.plot_circle(ax, 2 * np.sqrt(3), "g")
        self.plot_circle(ax, 4, "b")
        
        # 关闭坐标轴并显示图形
        plt.axis("off")
        plt.show()

这段代码的关键在于生成一个表示二维晶格原子位置的点集。我们逐行来详细解释：

  1. self.atoms.extend([])：extend是一个 Python 列表方法，用于将另一个列表的所有元素添加到当前列表的末尾。此行的目的是将定义的原子位置集合添加到 self.atoms 这个列表中。

  2. [4 * x, 2 * y * np.sqrt(3)]：这是晶格中原子的一个位置。它的 x 坐标是4 * x，y 坐标是 2 * y * np.sqrt(3)。这个位置是基于一个矩形网格，但 y坐标乘以了np.sqrt(3)的因子，使得在y方向上的间距与x方向上的间距不同，从而形成了一个密集的二维晶格。

  3. [2 + 4 * x, 2 * y * np.sqrt(3)]：这是另一个原子位置。它的 x 坐标是 4 * x 向右偏移了2个单位。y坐标和前面的原子位置相同。

  4. [1 + 4 * x, np.sqrt(3) + 2 * y * np.sqrt(3)] 和 [3 + 4 * x, np.sqrt(3) + 2 * y * np.sqrt(3)]：这两个原子位置在 y 方向上都比前两个原子上移了 np.sqrt(3) 的距离，而在 x 方向上分别向左和向右偏移了 1 和 3 个单位。

这四个原子位置组成了一个单元格，在二维平面上形成一个密集的原子阵列。通过 for x in np.arange(-2, 3):和for y in np.arange(-2, 3):两个嵌套的循环，这个单元格被重复多次，从而生成了一个更大的区域上的原子位置集合。

简而言之，这段代码通过一系列计算生成了一个代表二维密集晶格的原子位置点集。

import numpy as np
import matplotlib.pyplot as plt

# Constants
A = 0.010323
SIGMA = 3.405
R_CUTOFF_1 = 7.0
R_CUTOFF_2 = 7.5
DELTA_R = 0.001
START_R = 3
END_R = 10

def L_J_potential(r):
    """Calculate L-J potential for a given distance r."""
    if r <= R_CUTOFF_1:
        return 4 * A * ( (SIGMA/r)**12 - (SIGMA/r)**6 )
    elif r <= R_CUTOFF_2:
        return -6.8102128 * 0.001 * (r-R_CUTOFF_2)**3 - 5.5640876 * 0.001 * (r-R_CUTOFF_2)**2
    else:
        return 0

def compute_energy(r):
    """Compute energy considering the nearest two shells."""
    r_2 = np.sqrt(3) * r
    return 6 * L_J_potential(r) + 6 * L_J_potential(r_2)

def find_min_energy_and_r():
    """Iterate over r values to find the minimum energy and its corresponding r value."""
    r_values = np.arange(START_R, END_R, DELTA_R)
    energies = [compute_energy(r) for r in r_values]

    r_min = r_values[np.argmin(energies)]
    E_min = min(energies)
    return r_min, E_min, r_values, energies

def plot_energy_vs_distance(r_values, energies):
    """Plot L-J potential energy against distance."""
    plt.plot(r_values, energies)
    plt.xlabel('distance/Angstrom')
    plt.ylabel('L-J potential energy/eV')
    plt.title('the nearest two shells')
    plt.show()

def main():
    r_min, E_min, r_values, energies = find_min_energy_and_r()
    plot_energy_vs_distance(r_values, energies)
    print(E_min, r_min)

if __name__ == "__main__":
    main()

import numpy as np                 # 引入NumPy库，用于数值计算
import matplotlib.pyplot as plt   # 引入matplotlib库，用于绘图

# Constants
# 定义常量
A = 0.010323                      # Lennard-Jones势能的A值
SIGMA = 3.405                     # Lennard-Jones势能的σ值
R_CUTOFF_1 = 7.0                  # 第一个距离截断值
R_CUTOFF_2 = 7.5                  # 第二个距离截断值
DELTA_R = 0.001                  # r值的增量
START_R = 3                      # 开始的r值
END_R = 10                       # 结束的r值

def L_J_potential(r):
    """Calculate L-J potential for a given distance r."""
    # 根据给定的距离r计算L-J势能
    if r <= R_CUTOFF_1:
        # 如果r小于等于第一个截断值
        return 4 * A * ( (SIGMA/r)**12 - (SIGMA/r)**6 )
    elif r <= R_CUTOFF_2:
        # 如果r在两个截断值之间
        return -6.8102128 * 0.001 * (r-R_CUTOFF_2)**3 - 5.5640876 * 0.001 * (r-R_CUTOFF_2)**2
    else:
        # 如果r大于第二个截断值
        return 0

def compute_energy(r):
    """Compute energy considering the nearest two shells."""
    # 计算考虑最近的两个shells的能量
    r_2 = np.sqrt(3) * r            # 计算第二个shell的r值
    return 6 * L_J_potential(r) + 6 * L_J_potential(r_2)

def find_min_energy_and_r():
    """Iterate over r values to find the minimum energy and its corresponding r value."""
    # 遍历r值以找到最小的能量及其对应的r值
    r_values = np.arange(START_R, END_R, DELTA_R)     # 创建一个r值的数组

    E_min = min(energies)                             # 找到最小的能量值
    return r_min, E_min, r_values, energies

def plot_energy_vs_distance(r_values, energies):

    # 绘制L-J势能与距离的关系图
    plt.plot(r_values, energies)                      # 绘制能量与距离的图
    plt.xlabel('distance/Angstrom')                   # 设置x轴标签
    plt.ylabel('L-J potential energy/eV')             # 设置y轴标签
    plt.title('the nearest two shells')               # 设置图表标题
    plt.show()                                       # 显示图表

def main():
    # 主函数
    r_min, E_min, r_values, energies = find_min_energy_and_r()  # 获取最小能量、对应的r值和所有r值与能量的列表
    plot_energy_vs_distance(r_values, energies)                 # 绘制L-J势能与距离的图表
    print(E_min, r_min)                                         # 打印最小的能量值及其对应的r值

if __name__ == "__main__":
    # 如果此脚本作为主程序运行，则执行main()函数
